// 시리얼 수신 처리 함수 async function readSerialData(port) { const textDecoder = new TextDecoderStream(); port.readable.pipeTo(textDecoder.writable); const reader = textDecoder.readable.getReader(); let buffer = ""; try { while (true) { const { value, done } = await reader.read(); if (done) break; if (value) { buffer += value; // 더비 매직 데이터 패턴 매칭 (예: 1=2.9795 또는 A=2.9795) // 형식: [레인번호 또는 알파벳]=[기록] const regex = /([1-4A-D])\s*=\s*([0-9]+\.[0-9]+)/g; let match; while ((match = regex.exec(buffer)) !== null) { const rawLane = match[1]; const time = parseFloat(match[2]).toFixed(3); // 소수점 3자리 정리 // 레인 인덱스 계산 (1~4 또는 A~D -> 1~4) let laneNum = parseInt(rawLane); if (isNaN(laneNum)) { laneNum = rawLane.charCodeAt(0) - 64; // A->1, B->2, C->3, D->4 } // 해당 레인 입력 필드에 값 반영 updateLaneResult(laneNum, time); } // 버퍼가 너무 길어지면 최근 100글자만 유지 if (buffer.length > 200) { buffer = buffer.slice(-100); } } } } catch (err) { console.error("수신 오류:", err); } finally { reader.releaseLock(); } } // 화면 업데이트 함수 예시 function updateLaneResult(laneNum, time) { const input = document.getElementById(`record-lane-${laneNum}`); if (input) { input.value = time; input.classList.add('bg-yellow-100'); // 기록 들어왔을 때 강조 깜빡임 setTimeout(() => input.classList.remove('bg-yellow-100'), 1000); console.log(`[반영 성공] ${laneNum}번 레인: ${time}초`); } }
